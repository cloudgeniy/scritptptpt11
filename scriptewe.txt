local highlights, nameTags, boxes, skeletons = {}, {}, {}, {}

local function clearESP()
    for _, hl in pairs(highlights) do pcall(function() hl:Destroy() end) end
    for _, t in pairs(nameTags) do pcall(function() t:Destroy() end) end
    for _, b in pairs(boxes) do pcall(function() b:Destroy() end) end
    for _, s in pairs(skeletons) do if s then for _, line in pairs(s) do pcall(function() line:Destroy() end) end end end
    highlights, nameTags, boxes, skeletons = {}, {}, {}, {}
end

local function styleNameTag(lab, player)
    if S.nameStyle == "Default" then
        lab.TextColor3 = Color3.fromRGB(255,255,255); lab.TextStrokeTransparency = 0.3; lab.BackgroundTransparency = 1
    elseif S.nameStyle == "Simple" then
        lab.TextColor3 = Color3.fromRGB(200,200,200); lab.TextStrokeTransparency = 1; lab.BackgroundTransparency = 1
    elseif S.nameStyle == "Boxed" then
        lab.TextColor3 = Color3.fromRGB(255,255,255); lab.TextStrokeTransparency = 1; lab.BackgroundTransparency = 0.5; lab.BackgroundColor3 = Color3.fromRGB(0,0,0)
    elseif S.nameStyle == "Colored" then
        local r = math.abs(math.sin(tick() + player.UserId))
        lab.TextColor3 = Color3.fromRGB(math.floor(255 * r), math.floor(255 * (1-r)), 255); lab.TextStrokeTransparency = 0.3; lab.BackgroundTransparency = 1
    end
end

local function applyESP(player)
    if unloaded or player == LocalPlayer or not player.Character or not S.espEnabled then return end
    if highlights[player] then pcall(function() highlights[player]:Destroy() end) end
    if nameTags[player] then pcall(function() nameTags[player]:Destroy() end) end
    if boxes[player] then pcall(function() boxes[player]:Destroy() end) end
    if skeletons[player] then for _, l in pairs(skeletons[player]) do pcall(function() l:Destroy() end) end end
    local char = player.Character
    if S.espType == "Highlight" or S.espType == "Chams" then
        local hl = Instance.new("Highlight")
        hl.FillColor = S.espColor; hl.OutlineColor = Color3.fromRGB(255,255,255)
        hl.FillTransparency = S.espType == "Chams" and 0.1 or S.espTransparency
        hl.OutlineTransparency = 0.2
        hl.Parent = char; highlights[player] = hl
    end
    if S.espType == "Box" then
        local box = Instance.new("BoxHandleAdornment")
        box.Adornee = char:FindFirstChild("HumanoidRootPart") or char
        box.Size = Vector3.new(3, 5, 2)
        box.Color3 = S.espColor; box.Transparency = S.espTransparency
        box.AlwaysOnTop = true; box.ZIndex = 10; box.Parent = char
        boxes[player] = box
    end
    if S.espType == "Skeleton" then
        local lines = {}
        for i = 1, 6 do
            local line = Instance.new("LineHandleAdornment")
            line.Color3 = S.espColor; line.Thickness = 1.5; line.Transparency = 0.3
            line.AlwaysOnTop = true
            line.Adornee = char:FindFirstChild("HumanoidRootPart") or char
            line.Parent = char
            table.insert(lines, line)
        end
        skeletons[player] = lines
    end
    if S.espNames then
        local bill = Instance.new("BillboardGui")
        bill.Adornee = char:FindFirstChild("Head") or char
        bill.Size = UDim2.new(0,160,0,30); bill.StudsOffset = Vector3.new(0,2,0); bill.AlwaysOnTop = true
        local lab = Instance.new("TextLabel")
        lab.Size = UDim2.new(1,0,1,0); lab.BackgroundTransparency = 1
        lab.TextColor3 = Color3.fromRGB(255,255,255); lab.TextStrokeTransparency = 0.3
        lab.Font = Enum.Font.GothamBold; lab.TextSize = 12; lab.Parent = bill
        bill.Parent = char; nameTags[player] = bill
        styleNameTag(lab, player)
    end
end

local function refreshESP()
    if unloaded then return end
    clearESP()
    if not S.espEnabled then return end
    for _, p in ipairs(Players:GetPlayers()) do applyESP(p) end
end

local function updateNames()

    for player, bill in pairs(nameTags) do

            styleNameTag(bill.TextLabel, player)

end

-- ========== AIMBOT ==========
local function getTarget()
    if unloaded or not S.aimEnabled then return nil end
    local closest, cd = nil, S.aimFOV
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local hum = p.Character:FindFirstChild("Humanoid")
            if hum and hum.Health > 0 then
                local partName = aimPartMap[S.aimPart] or "Head"
                local part = p.Character:FindFirstChild(partName) or p.Character:FindFirstChild("HumanoidRootPart")
                if part then
                    local pos, on = Camera:WorldToViewportPoint(part.Position)
                    if on then
                        local center = Camera.ViewportSize / 2
                        local dx, dy = pos.X - center.X, pos.Y - center.Y
                        local d = math.sqrt(dx*dx + dy*dy)
                        if d < cd then cd = d; closest = part end
                    end
                end
            end
        end
    end
    return closest
end

local function doAimbot()
    if unloaded or not S.aimEnabled then return end
    if not UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then return end
    local t = getTarget()
    if t then Camera.CFrame = Camera.CFrame:Lerp(CFrame.new(Camera.CFrame.Position, t.Position), S.aimSmooth) end
end

-- ========== DRAWINGS ==========
local function updateDrawings()
    if unloaded then return end
    local vp = Camera.ViewportSize
    local center = Vector2.new(vp.X / 2, vp.Y / 2)
    if fovCircle then
        fovCircle.Position = center
        fovCircle.Radius = S.aimFOV
        fovCircle.Visible = S.aimCircle
    end
    if #crosshairLines >= 4 then
        local gap = 4
        local size = S.crosshairSize
        crosshairLines[1].From = Vector2.new(center.X - size - gap, center.Y); crosshairLines[1].To = Vector2.new(center.X - gap, center.Y)
        crosshairLines[2].From = Vector2.new(center.X + gap, center.Y); crosshairLines[2].To = Vector2.new(center.X + size + gap, center.Y)
        crosshairLines[3].From = Vector2.new(center.X, center.Y - size - gap); crosshairLines[3].To = Vector2.new(center.X, center.Y - gap)
        crosshairLines[4].From = Vector2.new(center.X, center.Y + gap); crosshairLines[4].To = Vector2.new(center.X, center.Y + size + gap)
        for _, line in ipairs(crosshairLines) do line.Visible = S.crosshairEnabled end
    end
    if S.espTracers and Drawing and Drawing.new then
        local activePlayers = {}
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("Head") then
                local pos, on = Camera:WorldToViewportPoint(p.Character.Head.Position)
                if on then table.insert(activePlayers, Vector2.new(pos.X, pos.Y)) end
            end
        end
        for i, targetPos in ipairs(activePlayers) do
            if not tracerLines[i] then
                pcall(function()
                    local line = Drawing.new("Line")
                    line.Thickness = 1; line.Color = S.tracersColor; line.Transparency = 0.8
                    table.insert(tracerLines, line)
                end)
            end
            if tracerLines[i] then
                tracerLines[i].From = Vector2.new(center.X, vp.Y)
                tracerLines[i].To = targetPos
                tracerLines[i].Visible = true
            end
        end
        for i = #activePlayers + 1, #tracerLines do
            if tracerLines[i] then tracerLines[i].Visible = false end
        end
    else
        for _, line in ipairs(tracerLines) do if line then line.Visible = false end end
    end
end

    if unloaded then return end
    if hitboxCon then hitboxCon:Disconnect() end

        for _, p in ipairs(Players:GetPlayers()) do

    origSizes = {}
end

-- ========== FLY ==========
local flying, flyBV, flyBG, flyCon = false, nil, nil, nil
local function startFly()
    if unloaded then return end
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChild("Humanoid")
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not (char and hum and root) then return end
    flying = true; hum.PlatformStand = true
    flyBV = Instance.new("BodyVelocity"); flyBV.MaxForce = Vector3.new(1e6,1e6,1e6); flyBV.Parent = root
    flyBG = Instance.new("BodyGyro"); flyBG.MaxTorque = Vector3.new(1e6,1e6,1e6); flyBG.Parent = root
    if flyCon then flyCon:Disconnect() end
    flyCon = RunService.RenderStepped:Connect(function()
        if unloaded or not flying then return end
        local spd = S.fastFlyRage and 400 or S.flySpeed
        local move = Vector3.new()
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then move = move + Camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then move = move - Camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then move = move - Camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then move = move + Camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then move = move + Vector3.new(0,1,0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then move = move - Vector3.new(0,1,0) end
        flyBV.Velocity = move * spd
        flyBG.CFrame = CFrame.new(root.Position, root.Position + Camera.CFrame.LookVector)
    end)
end
local function stopFly()
    flying = false
    local hum = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid")
    if hum then hum.PlatformStand = false end
    if flyBV then flyBV:Destroy() end
    if flyBG then flyBG:Destroy() end
    if flyCon then flyCon:Disconnect() end
end
local function toggleFly() if S.flyEnabled then startFly() else stopFly() end end

-- ========== FREECAM ==========
local freecamActive, freecamPos, freecamRot, freecamCon = false, nil, nil, nil
local function startFreeCam()
    if unloaded or freecamActive then return end
    freecamActive = true
    freecamPos = Camera.CFrame.Position
    freecamRot = Camera.CFrame
    if freecamCon then freecamCon:Disconnect() end
    freecamCon = RunService.RenderStepped:Connect(function()
        if unloaded or not freecamActive then return end
        local move = Vector3.new()
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then move = move + freecamRot.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then move = move - freecamRot.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then move = move - freecamRot.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then move = move + freecamRot.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then move = move + Vector3.new(0,1,0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then move = move - Vector3.new(0,1,0) end
        freecamPos = freecamPos + move * (S.freecamSpeed / 60)
        Camera.CFrame = CFrame.new(freecamPos) * (freecamRot - freecamRot.Position)
    end)
end
local function stopFreeCam() freecamActive = false; if freecamCon then freecamCon:Disconnect() end end
local function toggleFreeCam() if S.freecamEnabled then startFreeCam() else stopFreeCam() end end

    pcall(function() TeleportService:TeleportToPlaceInstance(game.PlaceId, game.JobId, LocalPlayer) end)

local function serverHop()
    pcall(function()
        local url = "https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100"
        local response = HttpService:JSONDecode(game:HttpGet(url))
        for _, server in pairs(response.data) do
            if server.id ~= game.JobId and server.playing < server.maxPlayers then
                TeleportService:TeleportToPlaceInstance(game.PlaceId, server.id, LocalPlayer)
                return
            end
        end
    end)
end

local function addSelector(parent, text, options, getter, setter, yPos)
    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(1, -10, 0, 60)
    frame.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
    frame.BorderSizePixel = 0
    frame.Parent = parent
    local c = Instance.new("UICorner"); c.CornerRadius = UDim.new(0, 10); c.Parent = frame
    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(1, -20, 0, 20)
    label.Position = UDim2.new(0, 10, 0, 5)
    label.BackgroundTransparency = 1
    label.Text = text; label.TextColor3 = Color3.fromRGB(200,200,200)
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Font = Enum.Font.GothamBold; label.TextSize = 12
    label.Parent = frame
    local btnContainer = Instance.new("Frame")
    btnContainer.Size = UDim2.new(1, -20, 0, 30)
    btnContainer.Position = UDim2.new(0, 10, 0, 25)
    btnContainer.BackgroundTransparency = 1
    btnContainer.Parent = frame
    local btnWidth = 1 / #options
    local buttons = {}
    for i, opt in ipairs(options) do
        local btn = Instance.new("TextButton")
        btn.Size = UDim2.new(btnWidth, -4, 1, 0)
        btn.Position = UDim2.new((i-1) * btnWidth, 2, 0, 0)
        btn.BackgroundColor3 = (getter() == opt) and Color3.fromRGB(200, 200, 200) or Color3.fromRGB(45, 45, 45)
        btn.Text = opt
        btn.TextColor3 = (getter() == opt) and Color3.fromRGB(0, 0, 0) or Color3.fromRGB(200, 200, 200)
        btn.Font = Enum.Font.GothamBold; btn.TextSize = 10
        btn.Parent = btnContainer
        local bc = Instance.new("UICorner"); bc.CornerRadius = UDim.new(0, 6); bc.Parent = btn
        btn.MouseButton1Click:Connect(function()
            setter(opt)
            for _, b in ipairs(buttons) do
                b.BackgroundColor3 = (b.Text == opt) and Color3.fromRGB(200, 200, 200) or Color3.fromRGB(45, 45, 45)
                b.TextColor3 = (b.Text == opt) and Color3.fromRGB(0, 0, 0) or Color3.fromRGB(200, 200, 200)
            end
        end)
        table.insert(buttons, btn)
    end
end

local function unloadCheat()
    unloaded = true
    for k, v in pairs(S) do if type(v) == "boolean" then S[k] = false end end
    pcall(clearESP); pcall(stopFly); pcall(stopHitbox); pcall(stopNoClip); pcall(stopFreeCam)
    for _, con in ipairs({flyBV, flyBG, noclipCon, flyCon, infJumpCon, hitboxCon, freecamCon, antiAfkCon, clickTpCon, bunnyHopCon, autoJumpCon, antiFlingCon, butterflyCon, jitterCon, desyncCon, killAuraCon, rapidFireCon, autoBlockCon, featherCon, instantKillCon, spinReverseCon}) do
        pcall(function()
            if con and con.Disconnect then con:Disconnect() end
            if con and con.Destroy then con:Destroy() end
        end)
    end
    pcall(function() if fovCircle then fovCircle:Remove() end end)
    for _, l in ipairs(crosshairLines) do pcall(function() l:Remove() end) end
    for _, l in ipairs(tracerLines) do pcall(function() l:Remove() end) end
    pcall(function()
        Lighting.Ambient = origAmbient; Lighting.OutdoorAmbient = origOutdoor
        Lighting.Brightness = origBrightness; Lighting.FogEnd = origFogEnd; Lighting.FogStart = origFogStart
    end)
    pcall(function() workspace.Gravity = origGravity end)
    pcall(function()
        LocalPlayer.CameraMinZoomDistance = origCamMin
        LocalPlayer.CameraMaxZoomDistance = origCamMax
    end)
    pcall(function()
        for _, part in ipairs(workspace:GetDescendants()) do
            if part:IsA("BasePart") then part.LocalTransparencyModifier = 0 end
        end
    end)
    pcall(function()
        local hum = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid")
        if hum then hum.WalkSpeed = 16; hum.JumpPower = 50; hum.PlatformStand = false end
    end)
    pcall(function() if wmGui then wmGui:Destroy() end end)
    pcall(function() if mobileGui then mobileGui:Destroy() end end)
    pcall(function() if screenGui then screenGui:Destroy() end end)
    pcall(function() if loadGui then loadGui:Destroy() end end)
    pcall(function() if playerListGui then playerListGui:Destroy() end end)
    print("[Cloud Hub v4] Выгружен.")
end
