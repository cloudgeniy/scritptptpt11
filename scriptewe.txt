-- ========== ESP ==========
local highlights, nameTags, boxes, skeletonDrawings = {}, {}, {}, {}

local function clearESP()
    for _, h in pairs(highlights) do pcall(function() h:Destroy() end) end
    for _, t in pairs(nameTags) do pcall(function() t:Destroy() end) end
    for _, b in pairs(boxes) do pcall(function() b:Destroy() end) end
    for _, lines in pairs(skeletonDrawings) do
        for _, l in pairs(lines) do pcall(function() l:Remove() end) end
    end
    highlights, nameTags, boxes, skeletonDrawings = {}, {}, {}, {}
end

local function styleNameTag(lab)
    if S.nameStyle == "Minimal" then
        lab.TextColor3 = Color3.fromRGB(255,255,255); lab.TextStrokeTransparency = 0.4
    elseif S.nameStyle == "HP Bar" then
        lab.TextColor3 = Color3.fromRGB(255,255,255); lab.TextStrokeTransparency = 0.2
    elseif S.nameStyle == "Full Info" then
        lab.TextColor3 = Color3.fromRGB(225,225,230); lab.TextStrokeTransparency = 0.1
    end
end

local function applyESP(player)
    if unloaded or player == LocalPlayer or not player.Character or not S.espEnabled then return end
    if highlights[player] then pcall(function() highlights[player]:Destroy() end) end
    if nameTags[player] then pcall(function() nameTags[player]:Destroy() end) end
    if boxes[player] then pcall(function() boxes[player]:Destroy() end) end
    if skeletonDrawings[player] then
        for _, l in pairs(skeletonDrawings[player]) do pcall(function() l:Remove() end) end
    end

    local char = player.Character
    if S.espType == "Highlight" or S.espType == "Chams" then
        local hl = Instance.new("Highlight")
        hl.FillColor = C.accent
        hl.OutlineColor = C.accent
        hl.FillTransparency = S.espType == "Chams" and 0.1 or S.espTransparency
        hl.OutlineTransparency = 0.2
        hl.Parent = char; highlights[player] = hl
    end
    if S.espType == "Box" then
        local box = Instance.new("BoxHandleAdornment")
        box.Adornee = char:FindFirstChild("HumanoidRootPart") or char
        box.Size = Vector3.new(3, 5, 2)
        box.Color3 = C.accent; box.Transparency = S.espTransparency
        box.AlwaysOnTop = true; box.ZIndex = 10; box.Parent = char
        boxes[player] = box
    end
    if S.espType == "Skeleton" and Drawing and Drawing.new then
        local lines = {}
        for i = 1, 10 do
            local l = Drawing.new("Line")
            l.Color = C.accent
            l.Thickness = 1.5
            l.Transparency = 0.75
            l.Visible = false
            table.insert(lines, l)
        end
        skeletonDrawings[player] = lines
    end
    if S.espNames then
        local bill = Instance.new("BillboardGui")
        bill.Adornee = char:FindFirstChild("Head") or char
        bill.Size = UDim2.new(0, 160, 0, 30)
        bill.StudsOffset = Vector3.new(0, 2.5, 0)
        bill.AlwaysOnTop = true
        local lab = Instance.new("TextLabel")
        lab.Size = UDim2.new(1, 0, 1, 0)
        lab.BackgroundTransparency = 1
        lab.Font = Enum.Font.GothamBold
        lab.TextSize = 12
        lab.Text = player.Name
        lab.Parent = bill
        bill.Parent = char; nameTags[player] = bill
        styleNameTag(lab)
    end
end

local function refreshESP()
    if unloaded then return end
    clearESP()
    if not S.espEnabled then return end
    for _, p in ipairs(Players:GetPlayers()) do applyESP(p) end
end

local function drawSkeleton(player, lines)
    local char = player.Character
    if not char then
        for _, l in ipairs(lines) do l.Visible = false end
        return
    end

    local head = char:FindFirstChild("Head")
    local root = char:FindFirstChild("HumanoidRootPart")
    local upperTorso = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
    local lowerTorso = char:FindFirstChild("LowerTorso")
    local lArm = char:FindFirstChild("LeftUpperArm") or char:FindFirstChild("Left Arm")
    local rArm = char:FindFirstChild("RightUpperArm") or char:FindFirstChild("Right Arm")
    local lLeg = char:FindFirstChild("LeftUpperLeg") or char:FindFirstChild("Left Leg")
    local rLeg = char:FindFirstChild("RightUpperLeg") or char:FindFirstChild("Right Leg")
    local lFoot = char:FindFirstChild("LeftFoot") or char:FindFirstChild("Left Leg")
    local rFoot = char:FindFirstChild("RightFoot") or char:FindFirstChild("Right Leg")

    if not (head and root) then
        for _, l in ipairs(lines) do l.Visible = false end
        return
    end

    local headP = head.Position
    local torsoP = upperTorso and upperTorso.Position or root.Position
    local lowerP = lowerTorso and lowerTorso.Position or root.Position
    local neckP = Vector3.new(headP.X, torsoP.Y, torsoP.Z)

    local function toScreen(pos)
        local v, on = Camera:WorldToViewportPoint(pos)
        if on and v.Z > 0 then return Vector2.new(v.X, v.Y) end
        return nil
    end

    local connections = {
        {headP, neckP},
        {neckP, torsoP},
        {torsoP, lArm and lArm.Position or torsoP},
        {torsoP, rArm and rArm.Position or torsoP},
        {torsoP, lowerP},
        {lowerP, lLeg and lLeg.Position or lowerP},
        {lowerP, rLeg and rLeg.Position or lowerP},
        {lLeg and lLeg.Position or lowerP, lFoot and lFoot.Position or lowerP},
        {rLeg and rLeg.Position or lowerP, rFoot and rFoot.Position or lowerP},
        {torsoP, root.Position},
    }

    for i, conn in ipairs(connections) do
        local a = toScreen(conn[1])
        local b = toScreen(conn[2])
        if a and b and lines[i] then
            lines[i].From = a
            lines[i].To = b
            lines[i].Visible = true
        elseif lines[i] then
            lines[i].Visible = false
        end
    end
end

local function updateSkeletons()
    if unloaded then return end
    for player, lines in pairs(skeletonDrawings) do
        drawSkeleton(player, lines)
    end
end

local function updateTracers()
    if unloaded then return end
    if not S.espTracers or not Drawing or not Drawing.new then
        for _, l in ipairs(tracerLines) do l.Visible = false end
        return
    end

    local vp = Camera.ViewportSize
    local bottom = Vector2.new(vp.X / 2, vp.Y)
    local active = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("Head") then
            local pos, on = Camera:WorldToViewportPoint(p.Character.Head.Position)
            if on and pos.Z > 0 then
                table.insert(active, Vector2.new(pos.X, pos.Y))
            end
        end
    end

    while #tracerLines < #active do
        local l = Drawing.new("Line")
        l.Thickness = 1
        l.Color = C.accent
        l.Transparency = 0.8
        l.Visible = false
        table.insert(tracerLines, l)
    end

    for i, target in ipairs(active) do
        tracerLines[i].From = bottom
        tracerLines[i].To = target
        tracerLines[i].Visible = true
    end
    for i = #active + 1, #tracerLines do
        tracerLines[i].Visible = false
    end
end

-- ========== AIMBOT ==========
local function getTargetPart(partName)
    if unloaded then return nil end
    local closest, cd = nil, S.aimFOV
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local hum = p.Character:FindFirstChild("Humanoid")
            if hum and hum.Health > 0 then
                local part = p.Character:FindFirstChild(aimPartMap[partName] or "Head") or p.Character:FindFirstChild("HumanoidRootPart")
                if part then
                    local pos, on = Camera:WorldToViewportPoint(part.Position)
                    if on then
                        local center = Camera.ViewportSize / 2
                        local dx, dy = pos.X - center.X, pos.Y - center.Y
                        local d = math.sqrt(dx*dx + dy*dy)
                        if d < cd then cd = d; closest = part end
                    end
                end
            end
        end
    end
    return closest
end

local originalMouseIndex, silentAimActive = nil, false

local function doSilentAim()
    if unloaded or not S.silentAim then
        if silentAimActive then
            pcall(function()
                local mouse = LocalPlayer:GetMouse()
                local mt = getrawmetatable(mouse)
                if mt and originalMouseIndex then
                    setreadonly(mouse, false)
                    mt.__index = originalMouseIndex
                    setreadonly(mouse, true)
                end
            end)
            silentAimActive = false
        end
        return
    end
    local t = getTargetPart(S.silentAimPart)
    if t then
        pcall(function()
            local mouse = LocalPlayer:GetMouse()
            local mt = getrawmetatable(mouse)
            if mt and mt.__index and not silentAimActive then
                originalMouseIndex = mt.__index
                local cf = CFrame.new(Camera.CFrame.Position, t.Position)
                setreadonly(mouse, false)
                mt.__index = function(self, key)
                    if key == "Hit" then return cf * CFrame.new(0, 0, -100) end
                    if key == "Target" then return t end
                    return originalMouseIndex(self, key)
                end
                setreadonly(mouse, true)
                silentAimActive = true
            end
        end)
    end
end

local function doAimbot()
    if unloaded or not S.aimEnabled then return end
    if not UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then return end
    local t = getTargetPart(S.aimPart)
    if t then
        local factor = math.clamp(S.aimSmooth, 0.01, 1)
        Camera.CFrame = Camera.CFrame:Lerp(CFrame.new(Camera.CFrame.Position, t.Position), factor)
    end
end

local flying, flyBV, flyBG, flyCon = false, nil, nil, nil
local function startFly()
    if unloaded then return end
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChild("Humanoid")
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not (char and hum and root) then return end
    flying = true; hum.PlatformStand = true
    flyBV = Instance.new("BodyVelocity"); flyBV.MaxForce = Vector3.new(1e6,1e6,1e6); flyBV.Parent = root
    flyBG = Instance.new("BodyGyro"); flyBG.MaxTorque = Vector3.new(1e6,1e6,1e6); flyBG.Parent = root
    if flyCon then flyCon:Disconnect() end
    flyCon = RunService.RenderStepped:Connect(function()
        if unloaded or not flying then return end
        local spd = S.fastFlyRage and 400 or S.flySpeed
        local move = Vector3.new()
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then move = move + Camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then move = move - Camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then move = move - Camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then move = move + Camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then move = move + Vector3.new(0,1,0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then move = move - Vector3.new(0,1,0) end
        flyBV.Velocity = move * spd
        flyBG.CFrame = CFrame.new(root.Position, root.Position + Camera.CFrame.LookVector)
    end)
end
local function stopFly()
    flying = false
    local hum = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid")
    if hum then hum.PlatformStand = false end
    if flyBV then flyBV:Destroy() end
    if flyBG then flyBG:Destroy() end
    if flyCon then flyCon:Disconnect() end
end
local function toggleFly() if S.flyEnabled then startFly() else stopFly() end end

local function addSlider(cont, label, minV, maxV, getter, setter, isDec)
    local box = Instance.new("Frame")
    box.Size = UDim2.new(1, 0, 0, 30)
    box.BackgroundTransparency = 1
    box.Parent = cont

    local info = Instance.new("Frame")
    info.Size = UDim2.new(1, 0, 0, 14)
    info.BackgroundTransparency = 1
    info.Parent = box

    local n = Instance.new("TextLabel")
    n.Size = UDim2.new(0.6, 0, 1, 0)
    n.BackgroundTransparency = 1
    n.Text = label
    n.TextColor3 = C.muted
    n.Font = Enum.Font.Gotham
    n.TextSize = 10
    n.TextXAlignment = Enum.TextXAlignment.Left
    n.Parent = info

    local v = Instance.new("TextLabel")
    v.Size = UDim2.new(0.4, 0, 1, 0)
    v.Position = UDim2.new(0.6, 0, 0, 0)
    v.BackgroundTransparency = 1
    v.Text = isDec and string.format("%.2f", getter()) or tostring(math.floor(getter()))
    v.TextColor3 = C.text
    v.Font = Enum.Font.Code
    v.TextSize = 10
    v.TextXAlignment = Enum.TextXAlignment.Right
    v.Parent = info

    local track = Instance.new("TextButton")
    track.Size = UDim2.new(1, 0, 0, 4)
    track.Position = UDim2.new(0, 0, 1, -6)
    track.BackgroundColor3 = C.cardHover
    track.Text = ""
    track.AutoButtonColor = false
    track.Parent = box
    local tc = Instance.new("UICorner"); tc.CornerRadius = UDim.new(1, 0); tc.Parent = track

    local fill = Instance.new("Frame")
    fill.Size = UDim2.new(math.clamp((getter() - minV) / (maxV - minV), 0, 1), 0, 1, 0)
    fill.BackgroundColor3 = C.accent
    fill.BorderSizePixel = 0
    fill.Parent = track
    local fc = Instance.new("UICorner"); fc.CornerRadius = UDim.new(1, 0); fc.Parent = fill

    local dragging = false
    local function upd(x)
        local rel = math.clamp((x - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
        local val = minV + rel * (maxV - minV)
        setter(val)
        v.Text = isDec and string.format("%.2f", val) or tostring(math.floor(val))
        fill.Size = UDim2.new(rel, 0, 1, 0)
    end
    track.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true; upd(i.Position.X)
        end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = false end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            upd(i.Position.X)
        end
    end)
end

RunService.RenderStepped:Connect(function()
    if unloaded then return end
    doAimbot()
    doSilentAim()
    updateSkeletons()
    updateTracers()

    local vp = Camera.ViewportSize
    local center = Vector2.new(vp.X / 2, vp.Y / 2)

    if fovCircle then
        fovCircle.Position = center
        fovCircle.Radius = S.aimFOV
        fovCircle.Visible = S.aimCircle and S.aimEnabled
    end

    if fovTargetLine then
        local tp = getTargetPart(S.aimPart)
        if tp and S.aimCircle and S.aimEnabled then
            local pos, on = Camera:WorldToViewportPoint(tp.Position)
            if on then
                fovTargetLine.From = center
                fovTargetLine.To = Vector2.new(pos.X, pos.Y)
                fovTargetLine.Visible = true
            else
                fovTargetLine.Visible = false
            end
        else
            fovTargetLine.Visible = false
        end
    end

    if #crosshairLines >= 4 then
        local gap = 4
        local sz = S.crosshairSize
        crosshairLines[1].From = Vector2.new(center.X - sz - gap, center.Y); crosshairLines[1].To = Vector2.new(center.X - gap, center.Y)
        crosshairLines[2].From = Vector2.new(center.X + gap, center.Y); crosshairLines[2].To = Vector2.new(center.X + sz + gap, center.Y)
        crosshairLines[3].From = Vector2.new(center.X, center.Y - sz - gap); crosshairLines[3].To = Vector2.new(center.X, center.Y - gap)
        crosshairLines[4].From = Vector2.new(center.X, center.Y + gap); crosshairLines[4].To = Vector2.new(center.X, center.Y + sz + gap)
        for _, l in ipairs(crosshairLines) do l.Visible = S.crosshairEnabled end
    end
end)

    local mbS = Instance.new("UIStroke"); mbS.Color = C.border; mbS.Transparency = 0.7; mbS.Parent = mobBtn

    local mbDrag, mbStart, mbPos, mbMoved = false, nil, nil, false
    mobBtn.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            mbDrag = true; mbMoved = false; mbStart = i.Position; mbPos = mobBtn.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if mbDrag and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            local d = i.Position - mbStart
            if math.abs(d.X) > 5 or math.abs(d.Y) > 5 then mbMoved = true end
            if mbMoved then
                local nx = mbPos.X.Offset + d.X
                local ny = mbPos.Y.Offset + d.Y
                local vp = Camera.ViewportSize
                nx = math.clamp(nx, 0, vp.X - mobBtn.AbsoluteSize.X)
                ny = math.clamp(ny, 0, vp.Y - mobBtn.AbsoluteSize.Y)
                mobBtn.Position = UDim2.new(0, nx, 0, ny)
            end
        end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            mbDrag = false
        end
    end)
    mobBtn.MouseButton1Click:Connect(function()
        if not mbMoved then
            S.guiVisible = not S.guiVisible
            if main then main.Visible = S.guiVisible end
        end
    end)
end

local function unloadCheat()
    unloaded = true
    for k, v in pairs(S) do if type(v) == "boolean" then S[k] = false end end
    pcall(clearESP); pcall(stopFly); pcall(stopHitbox); pcall(stopNoClip)
    pcall(function() if fovCircle then fovCircle:Remove() end end)
    pcall(function() if fovTargetLine then fovTargetLine:Remove() end end)
    for _, l in ipairs(crosshairLines) do pcall(function() l:Remove() end) end
    for _, l in ipairs(tracerLines) do pcall(function() l:Remove() end) end
    pcall(function()
        Lighting.Ambient = origAmbient
        Lighting.OutdoorAmbient = origOutdoor
        Lighting.Brightness = origBrightness
        Lighting.FogEnd = origFogEnd
        Lighting.FogStart = origFogStart
        workspace.Gravity = origGravity
        LocalPlayer.CameraMinZoomDistance = origCamMin
        LocalPlayer.CameraMaxZoomDistance = origCamMax
    end)
    pcall(function()
        local hum = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid")
        if hum then hum.WalkSpeed = 16; hum.JumpPower = 50; hum.PlatformStand = false end
    end)
    pcall(function() if screenGui then screenGui:Destroy() end end)
    pcall(function() if wmGui then wmGui:Destroy() end end)
    pcall(function() if hudGui then hudGui:Destroy() end end)
    pcall(function() if mobGui then mobGui:Destroy() end end)
    pcall(function() if plGui then plGui:Destroy() end end)
    print("[Cloud Hub v9.3] Выгружен.")
end
