local function getESPColor()
    local p = S.espColorPreset
    if p == "White" then return Color3.fromRGB(255,255,255) end
    if p == "Red" then return Color3.fromRGB(255,60,60) end
    if p == "Green" then return Color3.fromRGB(60,255,60) end
    if p == "Blue" then return Color3.fromRGB(60,140,255) end
    if p == "Purple" then return Color3.fromRGB(190,90,255) end
    if p == "Cyan" then return Color3.fromRGB(60,255,255) end
    if p == "Yellow" then return Color3.fromRGB(255,220,60) end
    if p == "Pink" then return Color3.fromRGB(255,120,200) end
    if p == "Rainbow" then
        return Color3.fromHSV((tick() * S.rainbowSpeed * 0.3) % 1, 1, 1)
    end
    return Color3.fromRGB(255,255,255)
end

local highlights, nameTags, boxes, skeletonDrawings, healthBars, headDots = {}, {}, {}, {}, {}, {}

local function clearESP()
    for _, t in pairs({highlights, nameTags, boxes, healthBars, headDots}) do
        for _, v in pairs(t) do pcall(function() v:Destroy() end) end
    end
    for _, lines in pairs(skeletonDrawings) do
        for _, l in pairs(lines) do pcall(function() l:Remove() end) end
    end
    highlights, nameTags, boxes, skeletonDrawings, healthBars, headDots = {}, {}, {}, {}, {}, {}
end

local function applyESP(player)
    if unloaded or player == LocalPlayer or not player.Character or not S.espEnabled then return end
    if S.espTeamCheck and player.Team == LocalPlayer.Team then return end
    if highlights[player] then pcall(function() highlights[player]:Destroy() end) end
    if nameTags[player] then pcall(function() nameTags[player]:Destroy() end) end
    if boxes[player] then pcall(function() boxes[player]:Destroy() end) end
    if skeletonDrawings[player] then for _, l in pairs(skeletonDrawings[player]) do pcall(function() l:Remove() end) end end
    if healthBars[player] then pcall(function() healthBars[player]:Destroy() end) end
    if headDots[player] then pcall(function() headDots[player]:Destroy() end) end

    local char = player.Character
    local col = getESPColor()

    if S.espType == "Highlight" or S.espType == "Chams" then
        local hl = Instance.new("Highlight")
        hl.FillColor = col
        hl.OutlineColor = col
        hl.FillTransparency = S.espType == "Chams" and 0.1 or S.espTransparency
        hl.OutlineTransparency = 0.2

    end
    if S.espType == "Box" then

end

local function drawSkeleton(player, lines)
    local char = player.Character
    if not char then
        for _, l in ipairs(lines) do l.Visible = false end
        return
    end
    local head = char:FindFirstChild("Head")
    local root = char:FindFirstChild("HumanoidRootPart")
    local ut = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
    local lt = char:FindFirstChild("LowerTorso")
    local la = char:FindFirstChild("LeftUpperArm") or char:FindFirstChild("Left Arm")
    local ra = char:FindFirstChild("RightUpperArm") or char:FindFirstChild("Right Arm")
    local ll = char:FindFirstChild("LeftUpperLeg") or char:FindFirstChild("Left Leg")
    local rl = char:FindFirstChild("RightUpperLeg") or char:FindFirstChild("Right Leg")
    local lf = char:FindFirstChild("LeftFoot") or char:FindFirstChild("Left Leg")
    local rf = char:FindFirstChild("RightFoot") or char:FindFirstChild("Right Leg")
    if not (head and root) then
        for _, l in ipairs(lines) do l.Visible = false end
        return
    end
    local hP = head.Position
    local tP = ut and ut.Position or root.Position
    local lP = lt and lt.Position or root.Position
    local nP = Vector3.new(hP.X, tP.Y, tP.Z)
    local function scr(p)
        local v, on = Camera:WorldToViewportPoint(p)
        if on and v.Z > 0 then return Vector2.new(v.X, v.Y) end
        return nil
    end
    local cn = {
        {hP, nP}, {nP, tP},
        {tP, la and la.Position or tP},
        {tP, ra and ra.Position or tP},
        {tP, lP},
        {lP, ll and ll.Position or lP},
        {lP, rl and rl.Position or lP},
        {ll and ll.Position or lP, lf and lf.Position or lP},
        {rl and rl.Position or lP, rf and rf.Position or lP},
        {tP, root.Position},
    }
    for i, c in ipairs(cn) do
        local a = scr(c[1]); local b = scr(c[2])
        if a and b and lines[i] then
            lines[i].From = a; lines[i].To = b; lines[i].Visible = true
        elseif lines[i] then
            lines[i].Visible = false
        end
    end
end

local function updateESPColors()

    local col = getESPColor()
    for _, h in pairs(highlights) do pcall(function() h.FillColor = col; h.OutlineColor = col end) end
    for _, b in pairs(boxes) do pcall(function() b.Color3 = col end) end
    for _, lines in pairs(skeletonDrawings) do
        for _, l in pairs(lines) do pcall(function() l.Color = col end) end
    end
    for _, d in pairs(headDots) do
        pcall(function() d.Frame.BackgroundColor3 = col end)
    end

    local vp = Camera.ViewportSize

-- AIMBOT
local function getTargetPart(partName)
    if unloaded then return nil end    local closest, cd = nil, S.aimFOV
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            if not (S.aimTeamCheck and p.Team == LocalPlayer.Team) then
                local hum = p.Character:FindFirstChild("Humanoid")
                if hum and hum.Health > 0 then
                    local part = p.Character:FindFirstChild(aimPartMap[partName] or "Head") or p.Character:FindFirstChild("HumanoidRootPart")
                    if part then
                        local blocked = false
                        if S.aimWallCheck then
                            local ray = Ray.new(Camera.CFrame.Position, (part.Position - Camera.CFrame.Position).Unit * 1000)
                            local hit = workspace:FindPartOnRay(ray, LocalPlayer.Character)
                            if hit and not hit:IsDescendantOf(p.Character) then blocked = true end
                        end
                        if not blocked then
                            local pos, on = Camera:WorldToViewportPoint(part.Position)
                            if on then
                                local center = Camera.ViewportSize / 2
                                local dx, dy = pos.X - center.X, pos.Y - center.Y
                                local d = math.sqrt(dx*dx + dy*dy)
                                if d < cd then cd = d; closest = part end
                            end
                        end
                    end
                end
            end
        end
    end
    return closest
end

local originalMouseIndex, silentAimActive = nil, false

local function doSilentAim()
    if unloaded or not S.silentAim then
        if silentAimActive then
            pcall(function()
                local mouse = LocalPlayer:GetMouse()
                local mt = getrawmetatable(mouse)
                if mt and originalMouseIndex then
                    setreadonly(mouse, false)
                    mt.__index = originalMouseIndex
                    setreadonly(mouse, true)
                end
            end)
            silentAimActive = false
        end
        return
    end
    local t = getTargetPart(S.silentAimPart)
    if t then
        pcall(function()
            local mouse = LocalPlayer:GetMouse()
            local mt = getrawmetatable(mouse)
            if mt and mt.__index and not silentAimActive then
                originalMouseIndex = mt.__index
                local cf = CFrame.new(Camera.CFrame.Position, t.Position)
                setreadonly(mouse, false)
                mt.__index = function(self, key)
                    if key == "Hit" then return cf * CFrame.new(0, 0, -100) end
                    if key == "Target" then return t end
                    return originalMouseIndex(self, key)
                end
                setreadonly(mouse, true)
                silentAimActive = true
            end
        end)
    end
end

    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not (char and hum and root) then return end

    flyBG = Instance.new("BodyGyro"); flyBG.MaxTorque = Vector3.new(1e6,1e6,1e6); flyBG.Parent = root
    if flyCon then flyCon:Disconnect() end
    flyCon = track(RunService.RenderStepped:Connect(function()

        local move = Vector3.new()
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then move = move + Camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then move = move - Camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then move = move - Camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then move = move + Camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then move = move + Vector3.new(0,1,0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then move = move - Vector3.new(0,1,0) end

    end))

    local tc = Instance.new("UICorner"); tc.CornerRadius = UDim.new(1, 0); tc.Parent = track

    fill.Size = UDim2.new(math.clamp((getter() - minV) / (maxV - minV), 0, 1), 0, 1, 0)
    fill.BackgroundColor3 = S.uiAccentColor
    fill.BorderSizePixel = 0
    fill.Parent = track
    local fc = Instance.new("UICorner"); fc.CornerRadius = UDim.new(1, 0); fc.Parent = fill
    registerTheme(function() track.BackgroundColor3 = S.uiCardColor; fill.BackgroundColor3 = S.uiAccentColor end)

    local dragging = false
    local function upd(x)
        local rel = math.clamp((x - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
        local val = minV + rel * (maxV - minV)
        setter(val)
        v.Text = isDec and string.format("%.2f", val) or tostring(math.floor(val))
        fill.Size = UDim2.new(rel, 0, 1, 0)
    end
    track.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true; upd(i.Position.X)
        end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = false end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            upd(i.Position.X)
        end
    end)
end
