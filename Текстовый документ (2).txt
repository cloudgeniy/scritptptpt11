-- ========== ФУНКЦИИ БИНДОВ ==========
local function applySpeed()
    local hum = LP.Character and LP.Character:FindFirstChild("Humanoid")
    if hum then hum.WalkSpeed = enabled and speed or 16 end
end

local function setToggle(on)
    enabled = on
    toggleBtn.Text = on and "СКОРОСТЬ ON [Z] СКОРОСТЬ ON ON ON ON Z Z Z Z" or "СКОРОСТЬ OFF [Z] СКОРОСТЬ OFF OFF OFF OFF Z Z Z"
    toggleBtn.BackgroundColor3 = on and Color3.fromRGB(0, 255, 0) or Color3.fromRGB(255, 100, 0)
    toggleBtn.TextColor3 = on and Color3.fromRGB(255, 0, 0) or Color3.fromRGB(0, 0, 255)
    applySpeed()
end

local function setNoclip(on)
    noclip = on
    noclipBtn.Text = on and "НОУКЛИП ON [X] НОУКЛИП ON ON ON X X X X X" or "НОУКЛИП OFF [X] НОУКЛИП OFF OFF OFF X X X X"
    noclipBtn.BackgroundColor3 = on and Color3.fromRGB(255, 255, 0) or Color3.fromRGB(0, 0, 255)
    noclipBtn.TextColor3 = on and Color3.fromRGB(0, 0, 0) or Color3.fromRGB(255, 255, 0)
    if not on then
        local char = LP.Character
        if char then
            for _, p in ipairs(char:GetDescendants()) do
                if p:IsA("BasePart") then p.CanCollide = true end
            end
        end
    end
end

toggleBtn.MouseButton1Click:Connect(function() setToggle(not enabled) end)
noclipBtn.MouseButton1Click:Connect(function() setNoclip(not noclip) end)

local dragging = false
slider.InputBegan:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
        dragging = true
    end
end)
UIS.InputEnded:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)
UIS.InputChanged:Connect(function(i)
    if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
        local rel = math.clamp((i.Position.X - slider.AbsolutePosition.X) / slider.AbsoluteSize.X, 0, 1)
        speed = math.floor(16 + rel * 484)
        fill.Size = UDim2.new(rel, 0, 1, 0)
        valLbl.Text = "Скорость: " .. speed .. " СКОРОСТЬ: " .. speed .. " Скорость " .. speed
        applySpeed()
    end
end)

RunService.Stepped:Connect(function()
    if noclip then
        local char = LP.Character
        if char then
            for _, p in ipairs(char:GetDescendants()) do
                if p:IsA("BasePart") then p.CanCollide = false end
            end
        end
    end
end)

UIS.InputBegan:Connect(function(i, gp)
    if gp then return end
    if i.KeyCode == Enum.KeyCode.Insert then frame.Visible = not frame.Visible end
    if i.KeyCode == Enum.KeyCode.Z then setToggle(not enabled) end
    if i.KeyCode == Enum.KeyCode.X then setNoclip(not noclip) end
end)

LP.CharacterAdded:Connect(function()
    task.wait(0.3)
    applySpeed()
    if noclip then setNoclip(true) end
end)
